# Analysis + ML pipeline: the versions that produced the published results.
#
# Pinned from the environment the thesis numbers were computed in, read from
# installed package metadata. The ML results are sensitive to the scikit-learn

#   python3.12 -m venv .venv
#   .venv/bin/pip install -r requirements/analysis.txt
#

matplotlib==3.9.4
